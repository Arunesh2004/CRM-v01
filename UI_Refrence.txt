/* ==========================================================================
   NEXUS CRM — Chart Factories (Chart.js)
   All builders are defensive: if the Chart.js CDN failed to load (offline,
   blocked network, ad-blocker), these quietly no-op instead of throwing —
   so a missing chart library can never break page rendering or navigation.
   ========================================================================== */
function chartLibReady(){ return typeof Chart !== 'undefined'; }

function chartDefaults(){
  if(!chartLibReady()) return;
  try{
    const dark = document.documentElement.classList.contains('dark');
    Chart.defaults.font.family = "'Inter', sans-serif";
    Chart.defaults.color = dark ? '#8891B0' : '#64748B';
    Chart.defaults.borderColor = dark ? 'rgba(255,255,255,.06)' : 'rgba(15,23,42,.06)';
  }catch(e){}
}

function buildRevenueChart(ctx){
  if(!chartLibReady() || !ctx) return null;
  try{
    chartDefaults();
    const grad = ctx.createLinearGradient(0,0,0,220);
    grad.addColorStop(0,'rgba(124,92,252,.35)');
    grad.addColorStop(1,'rgba(124,92,252,0)');
    return new Chart(ctx, {
      type:'line',
      data:{ labels:MOCK.charts.revenue.labels, datasets:[{
        label:'Revenue', data:MOCK.charts.revenue.data, borderColor:'#7C5CFC', backgroundColor:grad,
        borderWidth:3, fill:true, tension:.4, pointRadius:4, pointBackgroundColor:'#7C5CFC', pointBorderColor:'#fff', pointBorderWidth:2,
      }]},
      options:{ responsive:true, maintainAspectRatio:false, plugins:{legend:{display:false}, tooltip:{callbacks:{label:c=>fmtINR(c.raw)}}},
        scales:{ y:{ grid:{drawTicks:false}, ticks:{callback:v=>'₹'+(v/1000)+'k'} }, x:{ grid:{display:false} } } }
    });
  }catch(e){ console.warn('buildRevenueChart failed:', e); return null; }
}

/* ---------- Utilities: theme, dropdowns, toast, modal ---------- */
function toggleTheme(){
  const html = document.documentElement;
  html.classList.toggle('dark');
  try{ localStorage.setItem('nexus-theme', html.classList.contains('dark') ? 'dark' : 'light'); }catch(e){}
}
function safeIcons(){
  try{ if(window.lucide && typeof lucide.createIcons === 'function') lucide.createIcons(); }
  catch(e){ /* icon font failed to load (offline/blocked CDN) — app still fully functional */ }
}
function applyStoredTheme(){
  try{
    const stored = localStorage.getItem('nexus-theme');
    const prefersDark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;
    if(stored === 'dark' || (!stored && prefersDark)){
      document.documentElement.classList.add('dark');
    }
  }catch(e){ /* localStorage/matchMedia unavailable — default to light theme */ }
}
function toggleDropdown(id){
  document.querySelectorAll('.hidden-dropdown-managed').forEach(()=>{});
  const el = document.getElementById(id);
  const isHidden = el.classList.contains('hidden');
  document.querySelectorAll('[id$="Dropdown"]').forEach(d=>d.classList.add('hidden'));
  if(isHidden) el.classList.remove('hidden');
}
document.addEventListener('click', (e)=>{
  if(!e.target.closest('[onclick*="toggleDropdown"]') && !e.target.closest('[id$="Dropdown"]')){
    document.querySelectorAll('[id$="Dropdown"]').forEach(d=>d.classList.add('hidden'));
  }
});

function selectFolder(f){
  currentFolder = f;
  document.querySelectorAll('.folder-btn').forEach(b=>b.classList.toggle('active', b.dataset.folder===f));
  renderThreads();
}

function getFolderData(){
  if(currentFolder==='Sent') return MOCK.emails.sent.map(e=>({...e, from:'You', avatar:MOCK.currentUser.avatar, unread:false, starred:false}));
  if(currentFolder==='Drafts') return MOCK.emails.drafts.map(e=>({...e, from:'Draft', avatar:MOCK.currentUser.avatar, unread:false, starred:false}));
  if(currentFolder==='Starred') return MOCK.emails.threads.filter(t=>t.starred);
  if(currentFolder==='Spam' || currentFolder==='Trash' || currentFolder==='Scheduled') return [];
  return MOCK.emails.threads;
}

function declineCall(){
  document.getElementById('incomingCallOverlay').classList.add('hidden');
  showToast('Call declined','error');
}
function acceptCall(){
  document.getElementById('incomingCallOverlay').classList.add('hidden');
  document.getElementById('activeName').textContent = window.__pendingCall || 'Unknown';
  document.getElementById('activeCallOverlay').classList.remove('hidden');
  document.getElementById('activeCallOverlay').classList.add('flex');
  let secs = 0;
  callTimerInt = setInterval(()=>{
    secs++;
    const m = String(Math.floor(secs/60)).padStart(2,'0'), s = String(secs%60).padStart(2,'0');
    document.getElementById('callTimer').textContent = `${m}:${s}`;
  }, 1000);
  safeIcons();
}
function endCall(){
  clearInterval(callTimerInt);
  document.getElementById('activeCallOverlay').classList.add('hidden');
  showToast('Call ended — notes saved','success');
}

function buildMonth(){
  const days = ['Sun','Mon','Tue','Wed','Thu','Fri','Sat'];
  let html = days.map(d=>`<div class="text-center text-[10px] font-bold text-slate-400 py-2">${d}</div>`).join('');
  const firstDayOffset = 3; // Jul 1 2026 is Wednesday
  const totalDays = 31;
  for(let i=0;i<firstDayOffset;i++) html += `<div></div>`;
  for(let d=1; d<=totalDays; d++){
    const isToday = d===3;
    const evs = events[d] || [];
    html += `<div class="min-h-[76px] rounded-xl p-1.5 glass ${isToday?'ring-glow':''} cursor-pointer" onclick="showToast('Opening July ${d}','info')">
      <p class="text-[11px] font-medium ${isToday?'text-violet-500 font-bold':''}">${d}</p>

        ${evs.length>2 ? `<div class="text-[9px] text-slate-400">+${evs.length-2} more</div>` : ''}
      </div>
    </div>`;
  }
  document.getElementById('monthGrid').innerHTML = html;
}

function renderSidebarTopbar(pagekey, title){
  try{
    document.getElementById('sidebarMount').innerHTML = buildSidebar(pagekey);
    document.getElementById('topbarMount').innerHTML = buildTopbar(title);
  }catch(e){ console.error('Sidebar/topbar render failed:', e); }

  try{
    const cmdkTrigger = document.getElementById('cmdkTrigger');
    if(cmdkTrigger){
      cmdkTrigger.addEventListener('click', ()=>{
        const ov = document.getElementById('cmdkOverlay');
        if(ov){ ov.classList.remove('hidden'); document.getElementById('cmdkInput').focus(); }
      });
    }
  }catch(e){}

  try{
    const mobileBtn = document.getElementById('mobileMenuBtn');
    if(mobileBtn){
      mobileBtn.addEventListener('click', ()=>{
        document.getElementById('sidebar').classList.toggle('-translate-x-full');
        document.getElementById('sidebarBackdrop').classList.toggle('hidden');
      });
    }
  }catch(e){}

  try{
    const collapseBtn = document.getElementById('sidebarCollapseBtn');
    if(collapseBtn){
      collapseBtn.addEventListener('click', ()=>{
        const sb = document.getElementById('sidebar');
        sb.classList.toggle('lg:w-[76px]'); sb.classList.toggle('lg:w-64');
        document.querySelectorAll('.sidebar-label').forEach(l=>l.classList.toggle('lg:hidden'));
      });
    }
  }catch(e){}

  safeIcons();
}

function navigate(key){
  const page = PAGES[key];
  if(!page){ console.warn('Unknown page:', key); return; }
  try{
    document.getElementById('sidebar').classList.add('-translate-x-full');
    document.getElementById('sidebarBackdrop').classList.add('hidden');
    document.querySelectorAll('[id$="Dropdown"]').forEach(d=>d.classList.add('hidden'));
    document.getElementById('viewRoot').innerHTML = page.html;
    document.getElementById('viewRoot').scrollTop = 0;
  }catch(e){ console.error('View swap failed for', key, e); }

  renderSidebarTopbar(key, page.title);

  try{ page.init(); }
  catch(e){ console.error('Page init error for', key, e); }

  safeIcons();
  window.__currentPage = key;
}

function enterApp(){
  try{
    document.getElementById('authRoot').classList.add('hidden');
    document.getElementById('appRoot').classList.remove('hidden');
    document.getElementById('appRoot').classList.add('flex');
  }catch(e){ console.error('enterApp failed to toggle root visibility:', e); }
  navigate('dashboard');
}
function exitApp(){
  try{
    document.getElementById('appRoot').classList.add('hidden');
    document.getElementById('appRoot').classList.remove('flex');
    document.getElementById('authRoot').classList.remove('hidden');
  }catch(e){}
  showAuthView('login');
}

  try{
    document.getElementById('loginView').classList.toggle('hidden', view!=='login');
    document.getElementById('forgotView').classList.toggle('hidden', view!=='forgot-password');
  }catch(e){}
  safeIcons();
}

function initShellOnce(){
  try{ applyStoredTheme(); }catch(e){ console.warn('Theme init failed:', e); }

  try{ document.body.insertAdjacentHTML('beforeend', buildCommandPalette()); }
  catch(e){ console.warn('Command palette failed to build:', e); }

  try{ document.body.insertAdjacentHTML('beforeend', buildAIPanel()); }
  catch(e){ console.warn('AI panel failed to build:', e); }

  try{ document.body.insertAdjacentHTML('beforeend', '<div id="toastWrap" class="fixed bottom-6 left-6 z-[100] space-y-2"></div>'); }
  catch(e){ console.warn('Toast container failed to build:', e); }

  try{
    const cmdkOverlay = document.getElementById('cmdkOverlay');
    if(cmdkOverlay){
      cmdkOverlay.addEventListener('click', (e)=>{ if(e.target === cmdkOverlay) cmdkOverlay.classList.add('hidden'); });
      const cmdkInput = document.getElementById('cmdkInput');
      if(cmdkInput){
        cmdkInput.addEventListener('input', function(){
          const q = this.value.toLowerCase();
          document.querySelectorAll('#cmdkResults a').forEach(a=>{ a.style.display = a.textContent.toLowerCase().includes(q) ? 'flex' : 'none'; });
        });
      }
    }
  }catch(e){ console.warn('Command palette wiring failed:', e); }

  try{
    document.addEventListener('keydown', (e)=>{
      const cmdkOverlay = document.getElementById('cmdkOverlay');
      if(!cmdkOverlay) return;
      if((e.metaKey || e.ctrlKey) && e.key.toLowerCase()==='k'){ e.preventDefault(); cmdkOverlay.classList.remove('hidden'); const i=document.getElementById('cmdkInput'); if(i) i.focus(); }
      if(e.key === 'Escape') cmdkOverlay.classList.add('hidden');
    });
  }catch(e){}

  try{
    const backdrop = document.getElementById('sidebarBackdrop');
    if(backdrop){
      backdrop.addEventListener('click', ()=>{
        document.getElementById('sidebar').classList.add('-translate-x-full');
        backdrop.classList.add('hidden');
      });
    }
  }catch(e){}

  // THE CRITICAL PIECE: every <a href="something.html"> anywhere in the
  // document (sidebar, topbar dropdowns, command palette, or inside any
  // page's own markup) becomes an in-app SPA navigation instead of a real
  // page load. This MUST always register — it is deliberately the very
  // last thing in this function and has no dependency on anything above.
  try{
    document.addEventListener('click', function(e){
      const a = e.target.closest('a[href]');
      if(!a) return;
      const href = a.getAttribute('href');
      if(!href || !href.endsWith('.html')) return;
      e.preventDefault();
      const key = href.replace('.html','');
      if(key === 'login'){ exitApp(); }
      else if(key === 'forgot-password'){ showAuthView('forgot-password'); }
      else { navigate(key); }
    });
  }catch(e){ console.error('CRITICAL: link interceptor failed to register:', e); }

  safeIcons();
}

// Visible, non-blocking error banner. If anything unexpected throws
// anywhere in the app, the demo stays usable and the error is surfaced
// instead of failing completely silently.
window.addEventListener('error', function(e){
  try{
    if(document.getElementById('nexusErrorBanner')) return;
    const banner = document.createElement('div');
    banner.id = 'nexusErrorBanner';
    banner.style.cssText = 'position:fixed;bottom:16px;right:16px;z-index:99999;max-width:340px;background:#1B2340;color:#fff;padding:12px 16px;border-radius:12px;font:12px/1.4 sans-serif;box-shadow:0 8px 24px rgba(0,0,0,.35)';
    banner.textContent = 'A script on this demo page hit an unexpected error, but navigation should still work. (' + (e.message || 'unknown error') + ')';
    document.body.appendChild(banner);
    setTimeout(()=>banner.remove(), 6000);
  }catch(_){}
});
